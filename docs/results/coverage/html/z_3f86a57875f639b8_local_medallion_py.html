1"""Build the retail medallion in an in-memory DuckDB catalog from CSV files. 

2 

3Mirrors local-data-lakehouse's ``sql/retail/{bronze_ddl,silver_orders,gold_daily_metrics}.sql`` in DuckDB 

4SQL so contracts can be proven without Docker. The catalog is attached as ``lakehouse`` so contracts see 

5the same ``bronze/silver/gold`` schemas as on the real Iceberg catalog. 

6 

7Types follow what Spark writes to Iceberg: timestamps are ``timestamp with time zone``. 

8""" 

9 

10from __future__ import annotations 

11 

12from datetime import UTC, datetime 

13from pathlib import Path 

14from typing import Any 

15 

16from soda_lakehouse_dq.domain.errors import DataSourceError 

17 

18ORDER_FILES = "orders_*.csv" 

19CUSTOMER_FILE = "customers.csv" 

20 

21_BRONZE_ORDERS_DDL = """ 

22CREATE TABLE lakehouse.bronze.orders_raw (order_id VARCHAR, customer_id VARCHAR, order_ts TIMESTAMPTZ, 

23 status VARCHAR, amount DOUBLE, _source_file VARCHAR, _ingested_at TIMESTAMPTZ) 

24""" 

25 

26# One second apart per file so "latest _ingested_at wins" dedup behaves like successive Spark runs. 

27_BRONZE_ORDERS_INSERT = """ 

28INSERT INTO lakehouse.bronze.orders_raw 

29SELECT order_id, customer_id, CAST(order_ts AS TIMESTAMPTZ), status, CAST(amount AS DOUBLE), ?, 

30 CAST(? AS TIMESTAMPTZ) + to_seconds(?) 

31FROM read_csv(?, header = true, all_varchar = true) 

32""" 

33 

34_BRONZE_CUSTOMERS = """ 

35CREATE TABLE lakehouse.bronze.customers_raw AS 

36SELECT customer_id, name, email, city, CAST(signup_date AS DATE) AS signup_date, ? AS _source_file, 

37 CAST(? AS TIMESTAMPTZ) AS _ingested_at 

38FROM read_csv(?, header = true, all_varchar = true) 

39""" 

40 

41_SILVER_ORDERS = """ 

42CREATE TABLE lakehouse.silver.orders AS 

43SELECT order_id, customer_id, CAST(order_ts AS TIMESTAMPTZ) AS order_ts, LOWER(TRIM(status)) AS status, 

44 CAST(amount AS DOUBLE) AS amount, _source_file, _ingested_at 

45FROM (SELECT *, ROW_NUMBER() OVER (PARTITION BY order_id ORDER BY order_ts DESC, _ingested_at DESC) AS rn 

46 FROM lakehouse.bronze.orders_raw) 

47WHERE rn = 1 AND LOWER(TRIM(status)) IN ('paid', 'shipped', 'cancelled', 'refunded') 

48""" 

49 

50_SILVER_CUSTOMERS = """ 

51CREATE TABLE lakehouse.silver.customers AS 

52SELECT customer_id, TRIM(name) AS name, LOWER(TRIM(email)) AS email, TRIM(city) AS city, 

53 CAST(signup_date AS DATE) AS signup_date, _source_file, _ingested_at 

54FROM (SELECT *, ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY _ingested_at DESC) AS rn 

55 FROM lakehouse.bronze.customers_raw) 

56WHERE rn = 1 

57""" 

58 

59_GOLD = """ 

60CREATE TABLE lakehouse.gold.daily_order_metrics AS 

61SELECT CAST(order_ts AS DATE) AS order_date, CAST(COUNT(*) AS BIGINT) AS orders, 

62 SUM(CASE WHEN status IN ('paid', 'shipped') THEN amount ELSE 0 END) AS revenue, 

63 AVG(CASE WHEN status IN ('paid', 'shipped') THEN amount END) AS avg_order_value, 

64 CAST(SUM(CASE WHEN status = 'cancelled' THEN 1 ELSE 0 END) AS BIGINT) AS cancelled_orders, 

65 CAST(SUM(CASE WHEN status = 'refunded' THEN 1 ELSE 0 END) AS BIGINT) AS refunded_orders 

66FROM lakehouse.silver.orders 

67GROUP BY CAST(order_ts AS DATE) 

68ORDER BY order_date 

69""" 

70 

71 

72TABLES = ( 

73 "gold.daily_order_metrics", 

74 "silver.orders", 

75 "silver.customers", 

76 "bronze.orders_raw", 

77 "bronze.customers_raw", 

78) 

79

81def load_medallion(con: Any, sample_dir: Path, ingested_at: str | None = None) -> dict[str, int]: 

82 """(Re)build bronze, silver and gold in the catalog attached as ``lakehouse`` from CSV files. 

83 

84 Works on an in-memory catalog and on the live Iceberg REST catalog (DuckDB's iceberg extension 

85 supports CREATE SCHEMA / DROP TABLE / CREATE TABLE AS / INSERT). Returns row counts per table. 

86 """ 

87 orders = sorted(sample_dir.glob(ORDER_FILES)) 

88 customers = sample_dir / CUSTOMER_FILE 

89 if not orders or not customers.is_file(): 

90 raise DataSourceError(f"{sample_dir} needs {CUSTOMER_FILE} and {ORDER_FILES}") 

91 for schema in ("bronze", "silver", "gold"): 

92 con.execute(f"CREATE SCHEMA IF NOT EXISTS lakehouse.{schema}") 

93 for table in TABLES: 

94 con.execute(f"DROP TABLE IF EXISTS lakehouse.{table}") 

95 stamp = ingested_at or datetime.now(UTC).isoformat() 

96 con.execute(_BRONZE_ORDERS_DDL) 

97 for index, path in enumerate(orders): 

98 con.execute(_BRONZE_ORDERS_INSERT, [path.name, stamp, index, str(path)]) 

99 con.execute(_BRONZE_CUSTOMERS, [customers.name, stamp, str(customers)]) 

100 for statement in (_SILVER_ORDERS, _SILVER_CUSTOMERS, _GOLD): 

101 con.execute(statement) 

102 counts: dict[str, int] = {} 

103 for table in reversed(TABLES): 

104 row = con.execute(f"SELECT COUNT(*) FROM lakehouse.{table}").fetchone() # noqa: S608 (constant names) 

105 counts[table] = int(row[0]) if row else 0 

106 return counts 

107 

108 

109class LocalMedallionBuilder: 

110 """Callable connection factory: an in-memory DuckDB with bronze, silver and gold loaded.""" 

111 

112 def __init__(self, sample_dir: Path, ingested_at: str | None = None) -> None: 

113 self._dir = sample_dir 

114 self._ingested_at = ingested_at 

115 

116 def __call__(self) -> Any: 

117 import duckdb 

118 

119 con = duckdb.connect(":memory:") 

120 con.execute("SET TimeZone = 'UTC'") 

121 con.execute("ATTACH ':memory:' AS lakehouse") 

122 load_medallion(con, self._dir, self._ingested_at) 

123 return con
